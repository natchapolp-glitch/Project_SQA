 * A MultiValueMap decorates another map, allowing it to have
 * more than one value for a key.
 * <p>
 * A <code>MultiMap</code> is a Map with slightly different semantics.
 * Putting a value into the map will add the value to a Collection at that key.
 * Getting a value will return a Collection, holding all the values put to that key.
 * <p>
 * This implementation is a decorator, allowing any Map implementation

public class MultiValueMapTest {

    @Test
    public void testPutAndGetCollection() {
        MultiValueMap map = MultiValueMap.decorate(new HashMap());
        map.put("A", "1");
        map.put("A", "2");
        Collection coll = map.getCollection("A");
        assertNotNull(coll);
        assertEquals(2, coll.size());
        assertTrue(coll.contains("1"));
        assertTrue(coll.contains("2"));
        assertNull(map.getCollection("B"));
    }

    @Test
    public void testRemoveMapping() {
        MultiValueMap map = MultiValueMap.decorate(new HashMap());
        map.put("A", "1");
        map.put("A", "2");
        Object removed = map.removeMapping("A", "1");
        assertEquals("1", removed);
        assertEquals(1, map.size("A"));
        Object removedAgain = map.removeMapping("A", "999");
        assertNull(removedAgain);
        Object removedLast = map.removeMapping("A", "2");
        assertEquals("2", removedLast);
        assertFalse(map.containsKey("A"));
        assertNull(map.removeMapping("B", "x"));
    }

    @Test
    public void testContainsValue() {
        MultiValueMap map = MultiValueMap.decorate(new HashMap());
        map.put("A", "1");
        map.put("B", "2");
        assertTrue(map.containsValue("1"));
        assertTrue(map.containsValue("2"));
        assertFalse(map.containsValue("3"));
        assertTrue(map.containsValue("A", "1"));
        assertFalse(map.containsValue("A", "2"));
        assertFalse(map.containsValue("C", "1"));
    }

        assertEquals(2, map.size("A"));

        assertFalse(emptyResult);
