package org.apache.commons.jxpath.ri.model.dom;

import static org.junit.Assert.*;

import java.io.StringReader;
import java.util.Locale;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;

import org.apache.commons.jxpath.ri.QName;
import org.apache.commons.jxpath.ri.compiler.NodeTypeTest;
import org.apache.commons.jxpath.ri.Compiler;
import org.junit.Test;
import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.Node;
import org.w3c.dom.NodeList;
import org.xml.sax.InputSource;

public class DOMNodePointerTest {

    private Document parse(String xml) throws Exception {
        DocumentBuilderFactory dbf = DocumentBuilderFactory.newInstance();
        dbf.setNamespaceAware(true);
        DocumentBuilder db = dbf.newDocumentBuilder();
        return db.parse(new InputSource(new StringReader(xml)));
    }

    @Test
    public void testGetLocalNameNoPrefix() throws Exception {
        Document doc = parse("<root><child/></root>");
        Element child = (Element) doc.getDocumentElement().getFirstChild();
        assertEquals("child", DOMNodePointer.getLocalName(child));
        assertNull(DOMNodePointer.getPrefix(child));
    }

    @Test
    public void testGetLocalNameWithPrefix() throws Exception {
        Document doc = parse("<root xmlns:ns='urn:test'><ns:child/></root>");
        Element child = (Element) doc.getDocumentElement().getFirstChild();
        assertEquals("child", DOMNodePointer.getLocalName(child));
        assertEquals("ns", DOMNodePointer.getPrefix(child));
        assertEquals("urn:test", DOMNodePointer.getNamespaceURI(child));
    }

    @Test
    public void testGetNamespaceURINoNamespace() throws Exception {
        Document doc = parse("<root><child/></root>");
        Element child = (Element) doc.getDocumentElement().getFirstChild();
        assertNull(DOMNodePointer.getNamespaceURI(child));
    }

    @Test
    public void testTestNodeElementTypeTest() throws Exception {
        Document doc = parse("<root><child/></root>");
        Element child = (Element) doc.getDocumentElement().getFirstChild();
        NodeTypeTest test = new NodeTypeTest(Compiler.NODE_TYPE_NODE);
        assertTrue(DOMNodePointer.testNode(child, test));
    }

    @Test
    public void testTestNodeCommentTypeTest() throws Exception {
        Document doc = parse("<root><!--comment--></root>");
        Node comment = doc.getDocumentElement().getFirstChild();
        NodeTypeTest test = new NodeTypeTest(Compiler.NODE_TYPE_COMMENT);
        assertTrue(DOMNodePointer.testNode(comment, test));
        NodeTypeTest textTest = new NodeTypeTest(Compiler.NODE_TYPE_TEXT);
        assertFalse(DOMNodePointer.testNode(comment, textTest));
    }

    @Test
    public void testGetNameAndIsLeaf() throws Exception {
        Document doc = parse("<root><child/></root>");
        Element child = (Element) doc.getDocumentElement().getFirstChild();
        DOMNodePointer pointer = new DOMNodePointer(child, Locale.getDefault());
        QName name = pointer.getName();
        assertEquals("child", name.getName());
        assertTrue(pointer.isLeaf());
        assertFalse(pointer.isCollection());
    }

    @Test
    public void testCompareChildNodePointersOrder() throws Exception {
        Document doc = parse("<root><a/><b/><c/></root>");
        Element root = doc.getDocumentElement();
        NodeList children = root.getChildNodes();
        Node a = children.item(0);
        Node b = children.item(1);
        Node c = children.item(2);

        DOMNodePointer rootPointer = new DOMNodePointer(root, Locale.getDefault());
        DOMNodePointer pointerA = new DOMNodePointer(rootPointer, a);
        DOMNodePointer pointerB = new DOMNodePointer(rootPointer, b);
        DOMNodePointer pointerC = new DOMNodePointer(rootPointer, c);

        assertTrue(rootPointer.compareChildNodePointers(pointerA, pointerB) < 0);
        assertTrue(rootPointer.compareChildNodePointers(pointerB, pointerC) < 0);
        assertTrue(rootPointer.compareChildNodePointers(pointerC, pointerA) > 0);
        assertEquals(0, rootPointer.compareChildNodePointers(pointerA, pointerA));
    }
}

package org.apache.commons.jxpath.ri.model.jdom;

import static org.junit.Assert.*;

import java.util.Locale;

import org.apache.commons.jxpath.ri.QName;
import org.jdom.Document;
import org.jdom.Element;
import org.jdom.input.SAXBuilder;
import org.junit.Test;

public class JDOMNodePointerTest {

    private Document parse(String xml) throws Exception {
        SAXBuilder builder = new SAXBuilder();
        return builder.build(new StringReader(xml));
    }

    @Test
    public void testGetLocalNameNoPrefix() throws Exception {
        Document doc = parse("<root><child/></root>");
        Element child = (Element) doc.getRootElement().getContent().get(0);
