package org.apache.commons.compress.archivers.cpio;

import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;

import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.IOException;

import org.apache.commons.compress.archivers.ArchiveEntry;
import org.junit.Test;

public class CpioArchiveOutputStreamTest {

    @Test
    public void testConstructionAndFinish() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        CpioArchiveOutputStream out = new CpioArchiveOutputStream(baos, CpioConstants.FORMAT_NEW);
        out.finish();
        out.close();
        assertTrue(baos.size() > 0);
    }

    @Test
    public void testPutAndCloseArchiveEntry() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        CpioArchiveOutputStream out = new CpioArchiveOutputStream(baos);
        CpioArchiveEntry entry = new CpioArchiveEntry("testfile");
        entry.setFileSize(5);
        out.putArchiveEntry(entry);
        out.write("12345".getBytes());
        out.closeArchiveEntry();
        out.finish();
        out.close();
        assertTrue(baos.size() > 0);
    }

    @Test
    public void testCreateArchiveEntry() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        CpioArchiveOutputStream out = new CpioArchiveOutputStream(baos);
        File tempFile = File.createTempFile("compress", "tmp");
        tempFile.deleteOnExit();
        ArchiveEntry entry = out.createArchiveEntry(tempFile, "tempfile");
        assertNotNull(entry);
        out.close();
    }

    @Test(expected = IOException.class)
    public void testWriteAfterClose() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        CpioArchiveOutputStream out = new CpioArchiveOutputStream(baos);
        out.close();
        CpioArchiveEntry entry = new CpioArchiveEntry("test");
        out.putArchiveEntry(entry);
    }
}

package org.apache.commons.compress.archivers.zip;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;

import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.IOException;

import org.junit.Test;

public class ZipArchiveOutputStreamTest {

    @Test
    public void testOutputStreamConstructorAndFinish() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(baos);
        assertFalse(out.isSeekable());
        out.finish();
        out.close();
        assertTrue(baos.size() > 0);
    }

    @Test
    public void testFileConstructor() throws IOException {
        File tempFile = File.createTempFile("zipTest", ".zip");
        tempFile.deleteOnExit();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(tempFile);
        assertTrue(out.isSeekable());
        out.finish();
        out.close();
    }

    @Test
    public void testEncodingAndLanguageFlag() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(baos);
        out.setEncoding("UTF-8");
        assertEquals("UTF-8", out.getEncoding());
        out.setUseLanguageEncodingFlag(true);
        out.setFallbackToUTF8(true);
        out.finish();
        out.close();
    }

    @Test
    public void testPutStoredEntry() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(baos);
        ZipArchiveEntry entry = new ZipArchiveEntry("test.txt");
        entry.setMethod(ZipArchiveOutputStream.STORED);
        entry.setSize(4);
        entry.setCrc(0x3819d9b6L);
        out.putArchiveEntry(entry);
        out.write("test".getBytes());
        out.closeArchiveEntry();
        out.finish();
        out.close();
        assertTrue(baos.size() > 0);
    }

    @Test
    public void testSetCreateUnicodeExtraFields() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(baos);
        out.setCreateUnicodeExtraFields(ZipArchiveOutputStream.UnicodeExtraFieldPolicy.ALWAYS);
        out.finish();
        out.close();
    }

    @Test
    public void testCloseCreatesValidZipWhenEmpty() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipArchiveOutputStream out = new ZipArchiveOutputStream(baos);
        out.close();
        assertNotNull(baos.toByteArray());
    }
}

package org.apache.commons.compress.changes;

import static org.junit.Assert.assertNotNull;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;

import org.apache.commons.compress.archivers.cpio.CpioArchiveInputStream;
import org.apache.commons.compress.archivers.cpio.CpioArchiveOutputStream;
import org.junit.Test;

public class ChangeSetPerformerTest {

    @Test
    public void testConstructorAndPerformEmpty() throws IOException {
        ChangeSet set = new ChangeSet();
        ChangeSetPerformer performer = new ChangeSetPerformer(set);
        assertNotNull(performer);

        ByteArrayOutputStream emptyOut = new ByteArrayOutputStream();
        CpioArchiveOutputStream cpioOut = new CpioArchiveOutputStream(emptyOut);
        cpioOut.finish();
        cpioOut.close();

        ByteArrayInputStream in = new ByteArrayInputStream(emptyOut.toByteArray());
        CpioArchiveInputStream archiveIn = new CpioArchiveInputStream(in);

        ByteArrayOutputStream resultOut = new ByteArrayOutputStream();
        CpioArchiveOutputStream archiveOut = new CpioArchiveOutputStream(resultOut);

        ChangeSetResults results = performer.perform(archiveIn, archiveOut);
        assertNotNull(results);
        archiveIn.close();
        archiveOut.close();
    }

    @Test
    public void testAddChangeToSet() throws IOException {
        ChangeSet set = new ChangeSet();
        ByteArrayInputStream content = new ByteArrayInputStream("hello".getBytes());
        set.add("foo.txt", content);
        ChangeSetPerformer performer = new ChangeSetPerformer(set);
        assertNotNull(performer);
    }
}
